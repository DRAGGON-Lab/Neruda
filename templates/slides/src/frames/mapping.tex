\begin{frame}{The map scales the first component}
  \begin{equation}
    \nerudavector{y} = \nerudamatrix{A}\,\nerudavector{x}.
    \label{eq:scaling}
  \end{equation}
  \begin{description}[$\nerudamatrix{A}$]
    \item[$\nerudavector{y}$] Real two-component output column vector, in cm.
    \item[$\nerudamatrix{A}$] Dimensionless \(2\times2\) diagonal matrix
      with first entry \(2\) and second entry \(1\).
    \item[$\nerudavector{x}$] Real two-component input column vector, in cm.
  \end{description}
\end{frame}
